# To request a certificate from the service-ca clusteroperator, annotate the service for your app. This creates the secret
oc -n app1 annotate service/foo service.beta.openshift.io/serving-cert-secret-name=foo-secret

# Deploy trusted app
oc new-project app2
oc new-app --name bar quay.io/rdacosta/my_httpd:latest

# Create and annotate configmap
oc create cm ca-bundle
oc annotate configmap ca-bundle service.beta.openshift.io/inject-cabundle=true

# Patch trusting app to use the CA cert from service-ca
oc patch deploy/bar --patch-file patch_trusting_app.yaml
